def get_access_token():
    url = f"https://aip.baidubce.com/oauth/2.0/token?grant_type=client_credentials&client_id={API_KEY}&client_secret={SECRET_KEY}"
    try:
        res = requests.post(url)
        return res.json().get("access_token")
    except Exception as e:
        print("获取百度 Access Token 失败:", e)
        return None

def detect_emotion(image_path, token):
    if not token:
        return "neutral"
    url = f"https://aip.baidubce.com/rest/2.0/face/v3/detect?access_token={token}"
    try:
        with open(image_path, 'rb') as f:
            img_base64 = base64.b64encode(f.read()).decode()
        payload = {
            "image": img_base64,
            "image_type": "BASE64",
            "face_field": "emotion",
            "max_face_num": 1
        }
        res = requests.post(url, json=payload)
        data = res.json()
        if data.get("result") and data["result"].get("face_num", 0) > 0:
            return data["result"]["face_list"][0]["emotion"]["type"]
    except Exception as e:
        print("人脸表情检测请求异常:", e)
    return "neutral"

#  监听 Pico 串口消息 
def pico_listener():
    while True:
        if ser.in_waiting:
            try:
                line = ser.readline().decode().strip()
                print("收到 Pico:", line)

                if line == "ENTER_PLANET":
                    print("拍头，发送 pat_head")
                    socketio.emit('pat_head', {'data': 'tap'})

                elif line == "TOUCH_BACK":
                    print("背部触摸，发送 pat_back")
                    socketio.emit('pat_back', {'data': 'back'})

                elif line == "TOUCH_EAR":
                    print("耳朵触摸，发送 touch_ear")
                    socketio.emit('touch_ear', {'action': 'twinkle'})

                elif line == "SUMMON_FOX":
                    print("召唤狐狸")
                    socketio.emit('summon_fox', {})

                elif line.startswith("GESTURE_"):
                    gesture = line.replace("GESTURE_", "").lower()
                    print("手势:", gesture)
                    socketio.emit('gesture', {'value': gesture})

            except Exception as e:
                print("解析错误:", e)
        time.sleep(0.05)

#  情绪检测线程
def emotion_loop():
    picam2 = Picamera2()
    picam2.configure(picam2.create_preview_configuration(main={"size": (320, 240)}))
    picam2.start()
    time.sleep(1)

    token = get_access_token()
    print("开始检测情绪...")
    last = None

    while True:
        try:
            frame = picam2.capture_array()
            # 使用 PIL 保存图片，RGBA 转 RGB
            Image.fromarray(frame).convert('RGB').save("/tmp/face.jpg")
            emotion = detect_emotion("/tmp/face.jpg", token)

            if emotion != last:
                print("情绪:", emotion)

                pico_cmd = emotion_to_pico.get(emotion, b"NEUTRAL\n")
                ser.write(pico_cmd)
                print("  -> 发送给Pico:", pico_cmd)

                frontend_cmd = emotion_to_frontend.get(emotion, 'neutral')
                socketio.emit('emotion', {'value': frontend_cmd})
                print("  -> 发送给前端:", frontend_cmd)

                last = emotion
        except Exception as e:
            print("情绪循环异常:", e)

        time.sleep(2)
